#include <Arduino.h>
#include "MPX5010DP.h"
#include "SFM3019.h"
#include "Pid.h"

#define BLOWER_CONTROL_PIN    7
#define I2C_CLOCK_SPEED 400000UL
#define FLOW_FACTOR   1.2754F
#define MERIT1410_PIN   A7

float tidalVolume = 0;
uint32_t previousMillis;
uint8_t enable = 0;
uint8_t print_gains = 0;
uint8_t print = 0;

float ctrlRaw = 0, flowRef = 8.0, prevError = 0;
float pTerm = 0, iTerm = 0, dTerm = 0;
uint32_t timer, prevIterationTime;
float kp = 33.0, ki = 60.0, kd = 0.0;
uint8_t feedforward = 80;
//float kp = 3.8918, ki = 16.0769, kd = 0.2355;

uint32_t refTimer;
uint8_t flag = 0;
float flowMeas = 0;

void read_cmds(void);
void read_flow_sensor(void);
void blower_set_speed(uint8_t);
void update_flow_ref(void);
void debug(void);
void flow_control(float, float);
float map_float(float x, float in_min, float in_max, float out_min, float out_max);

PidController_t flowCtrl;

void setup() {
  Serial.begin(115200UL);

  Wire.begin();
  Wire.setClock(I2C_CLOCK_SPEED); //set the I2C clock speed to 400kHz

  pinMode(MERIT1410_PIN, INPUT);

  pinMode(BLOWER_CONTROL_PIN, OUTPUT);
  blower_set_speed(0);

  delay(1000);

  SFMInit();
  MPX5010_init();

  PidControllerInit(&flowCtrl);
  PidControllerSetGains(&flowCtrl, kp, ki, kd, 0.0, 0.8, 0.001);
  PidControllerSetLimits(&flowCtrl, 20, 0, 250, -250, 255, 0);
  PidControllerSetFFD(&flowCtrl, 80);
}

void loop() {

  if (millis() - timer >= 5) {
    timer = millis();

    read_cmds();
    //update_flow_ref();

    //read_flow_sensor();
    MPX5010_update_pressure();
    MPX5010_filtered_pressure();

    //ctrlRaw = PidControllerUpdate(&flowCtrl, flowRef, mpx.filteredPressure);
    // ctrlRaw = PidControllerUpdate(&flowCtrl, flowRef, FlowMeter.rawFlow);

    /**
     * kp = 33
     * ki = 60 ou 150
     * feedforward = 60 ou 80
     * Ts = 5ms
     * Filtro MV = 10 amostras
     */
    flow_control(flowRef, mpx.filteredPressure);

    if (enable) {
      blower_set_speed((uint8_t)ctrlRaw);
    } else {
      blower_set_speed(0);
    }

    if (print) debug();
  }  
}

void read_flow_sensor(void) {
  if (SFMReadSensor(&FlowMeter.rawFlow) == INITIALIZED) {

    flowMeas = (float)(FlowMeter.rawFlow * FLOW_FACTOR);
    
    if (abs(flowMeas) > 0.5) {
      tidalVolume += ((flowMeas / 60.0) * (millis() - previousMillis));
      previousMillis = millis();

      if (tidalVolume < 0) tidalVolume = 0;
    } else {
      tidalVolume = 0;
    }
  }
}


float limit_rate(float current, float target, float percent) {
  float delta = target - current;
  float maxDelta = percent * fabs(current); // usa valor absoluto

  if (delta > maxDelta) {
    return current + maxDelta;
  } else if (delta < -maxDelta) {
    return current - maxDelta;
  } else {
    return target;
  }
}

void read_cmds(void) {

  if (Serial.available() > 0) {
    char c = (char)Serial.read();

    if (c == '0') enable = !enable;
    if (c == 'c') MPX5010_set_offset(mpx.filteredPressure);
    if (c == 'n') feedforward += 5;
    if (c == 'm') feedforward -= 5;
    if (c == 'o') print = !print;
    if (c == 'p') print_gains = !print_gains;

    if (c == '1') flowRef += 1;
    if (c == '2') flowRef -= 1;

    if (c == '3') ctrlRaw += 10;
    if (c == '4') ctrlRaw -= 10;


    if (c == 'q') kp += 1;
    if (c == 'w') kp -= 1;
    if (c == 'a') ki += 1;
    if (c == 's') ki -= 1;
    if (c == 'z') kd += 0.05;
    if (c == 'x') kd -= 0.05;

    PidControllerSetKp(&flowCtrl, kp);
    PidControllerSetKi(&flowCtrl, ki);
    PidControllerSetKd(&flowCtrl, kd);
  }
}

void blower_set_speed(uint8_t pwmValue) {
  pwmValue = constrain(pwmValue, 0, 255);
  analogWrite(BLOWER_CONTROL_PIN, pwmValue);
}

  Serial.print(flowRef);

void flow_control(float desiredFlow, float feedback) {
  float sampleTime = (millis() - prevIterationTime) * 0.001;

  /*error*/
  float error = desiredFlow - feedback;

  float _kp = 1;
  _kp = map_float(abs(error), 0.5, 1.5, 33.0, 45.0);
  kp = _kp;

  if (kp < 33) kp = 33;
  if (kp > 45) kp = 45;

  /*proportional*/
  if (kp != 0) {
    pTerm = kp * error;
  } else {
    pTerm = 0;
  }

  /*integration*/
  if (ki != 0) {
    iTerm = iTerm + (ki * error * sampleTime);
    if (iTerm > 255) iTerm = 255;
    if (iTerm < 0) iTerm = 0;
  } else {
    iTerm = 0;
  }

  /*derivative*/
  if (kd != 0) {
    dTerm = kd * (error - prevError) / sampleTime;
  } else {
    dTerm = 0;
  }

  ctrlRaw = pTerm + iTerm + dTerm + feedforward;

  if (ctrlRaw > 255) ctrlRaw = 255;
  if (ctrlRaw < 0) ctrlRaw = 0;

  prevError = error;
  prevIterationTime = millis();
}

void update_flow_ref(void) {

  if (!flag && millis() - refTimer >= 5000) {
    flowRef = 30;
    flag = 1;
    refTimer = millis();
  } else if (flag && millis() - refTimer >= 5000) {
    flowRef = 50;
    flag = 0;
    refTimer = millis();
  }
}

float map_float(float x, float in_min, float in_max, float out_min, float out_max) {
  return (x - in_min) * (out_max - out_min) / (in_max - in_min) + out_min;
}
